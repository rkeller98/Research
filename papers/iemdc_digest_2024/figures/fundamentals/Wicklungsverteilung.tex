\begin{tikzpicture}[
]
    \pgfmathsetmacro{\factor}{3.25e-2}
    \pgfmathsetmacro{\N}{12}
    \pgfmathsetmacro{\p}{5}
    \pgfmathsetmacro{\poles}{2*\p}
    \pgfmathsetmacro{\ri}{60.2*\factor}
    \pgfmathsetmacro{\ro}{98.8*\factor}
    \pgfmathsetmacro{\rm}{0.5*(\ri+\ro)}
    \pgfmathsetmacro{\mcf}{0.84}
    \pgfmathsetmacro{\hpm}{8.2*\factor}
    \pgfmathsetmacro{\hyoke}{7.7*\factor}
    \pgfmathsetmacro{\g}{2*\factor}

    \pgfmathsetmacro{\hshoe}{5*\factor}
    \pgfmathsetmacro{\hcore}{25*\factor}
    \pgfmathsetmacro{\wcoil}{8*\factor}
    \pgfmathsetmacro{\wslot}{2*\factor}
    \pgfmathsetmacro{\wsec}{2*pi/\N*\rm }
    \pgfmathsetmacro{\wcore}{2*pi/\N*\rm - (2*\wcoil + \wslot)}
    \pgfmathsetmacro{\wshoe}{2*pi/\N*\rm - \wslot}

    \pgfmathsetmacro{\magSecWidth}{pi/\p*\rm}
    \pgfmathsetmacro{\wmag}{\magSecWidth*\mcf}
    \pgfmathsetmacro{\wmagair}{\magSecWidth*(1-\mcf)}

    \pgfmathsetmacro{\axlen}{\hcore/2 + \hshoe + \g + \hpm + \hyoke}

    \pgfmathsetmacro{\curveangle}{12.5}

    \pgfmathsetmacro{\fluxlen}{\axlen}

% architecture-local-helper: phaseArray - literal phase sequence used only in this winding drawing
    \newcommand{\phaseArray}{{1},{2},{-2},{-3},{3},{1},{-1},{-2},{2},{3},{-3},{-1}}
    % Poles
    \foreach \x in {1,...,\poles}
    {
        \pgfmathsetmacro{\xleft}{-\wmag/2 + \magSecWidth*(\x-5.5)}
        \ifthenelse{\isodd{\x}}
        {
            \draw[north magnet] (\xleft, {(\hcore/2 + \hshoe + \g + \hpm)}) rectangle ++(\wmag, {-\hpm});
            \draw[north magnet] (\xleft, {-(\hcore/2 + \hshoe + \g + \hpm)}) rectangle ++(\wmag, {+\hpm});
        }
        {
            \draw[south magnet] (\xleft, {(\hcore/2 + \hshoe + \g + \hpm)}) rectangle ++(\wmag, {-\hpm});
            \draw[south magnet] (\xleft, {-(\hcore/2 + \hshoe + \g + \hpm)}) rectangle ++(\wmag, {+\hpm});
        }
    }

    % Rotoryoke
    \draw[iron region] (-6*\wsec, {-(\hcore/2 + \hshoe + \g + \hpm)})
    rectangle ++(12*\wsec, -\hyoke)
    ;
    \draw[iron region] (-6*\wsec, {(\hcore/2 + \hshoe + \g + \hpm)})
    rectangle ++(12*\wsec, \hyoke)
    ;

    % Statorteeth
    \foreach \pha [count=\x] in \phaseArray
    {
        \draw[iron region] ({-\wcore/2 + \wsec*(\x-6.5)}, -\hcore/2)
            -- ++(0, \hcore) coordinate(P_NE_L)
            -- ++(-\wcoil, 0)
            -- ++(0, \hshoe)
            -- ++(\wshoe, 0)
            -- ++(0, -\hshoe) coordinate(P_NE_R)
            -- ++(-\wcoil, 0)
            -- ++(0, -\hcore) coordinate(P_SW_R)
            -- ++(\wcoil, 0)
            -- ++(0, -\hshoe)
            -- ++(-\wshoe, 0)
            -- ++(0, \hshoe) coordinate(P_SW_L)
            -- cycle
            ;

        \coordinate(center) at ($(P_SW_L)!0.5!(P_NE_R)$);
        \coordinate(lc) at ($(P_NE_L)!0.6!(P_SW_L)$);
        \coordinate(rc) at ($(P_NE_R)!0.4!(P_SW_R)$);

        \ifnum \pha = -3
            \path[winding positive, phase w quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding negative, phase w quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase w quantity] ($(0,\fluxlen/2) + (center)$) -- ++(0,-\fluxlen);
        \else\ifnum \pha = -2
            \path[winding positive, phase v quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding negative, phase v quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase v quantity] ($(0,\fluxlen/2) + (center)$) -- ++(0,-\fluxlen);
        \else\ifnum \pha = -1
            \path[winding positive, phase u quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding negative, phase u quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase u quantity] ($(0,\fluxlen/2) + (center)$) -- ++(0,-\fluxlen);
        \else\ifnum \pha = 1
            \path[winding negative, phase u quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding positive, phase u quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase u quantity] ($(0,-\fluxlen/2) + (center)$) -- ++(0,\fluxlen);
        \else\ifnum \pha = 2
            \path[winding negative, phase v quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding positive, phase v quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase v quantity] ($(0,-\fluxlen/2) + (center)$) -- ++(0,\fluxlen);
        \else\ifnum \pha = 3
            \path[winding negative, phase w quantity] (lc|-P_NE_L) -- (lc|-P_SW_L);
            \path[winding positive, phase w quantity] (rc|-P_NE_R) -- (rc|-P_SW_R);
            \draw[flux arrow, phase w quantity] ($(0,-\fluxlen/2) + (center)$) -- ++(0,\fluxlen);
        \fi\fi\fi\fi\fi\fi

        \node[background region, inner sep=2pt] at (center) {\x};
    }
\end{tikzpicture}
